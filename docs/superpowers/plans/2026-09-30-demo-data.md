# Demo Data (batch CV intake + reset) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the Isa demo a repeatable loop — reset to an empty operational database, load the whole `cvs/` corpus through the real ingestion endpoint, watch live counts, reset again.

**Architecture:** A new local-only, recruiter-only `/demo` API pair (`POST /demo/reset`, `GET /demo/status`) plus a new `/demo` page. Uploads reuse the existing `POST /documents` endpoint from the browser with concurrency 4. Reset is one transaction of FK-ordered bulk deletes that keeps `users` and the append-only `audit_events`, then writes one `demo_reset` audit event.

**Tech Stack:** FastAPI, SQLAlchemy 2, pytest + TestClient; Next.js 15, React 19, Vitest, Tailwind 4.

## Global Constraints

- Do NOT commit or push anything. Checkpoints replace commit steps.
- Do NOT modify `/intake` (CV Intake) or the ingestion pipeline. No bulk endpoint, no server-side folder ingest.
- All counts shown in the UI come from API responses. Never hard-code corpus outcome numbers.
- Demo routes exist only when `settings.environment == "local"`. Recruiter role required; engineer gets 403.
- Audit stays append-only. Reset detail uses only `ALLOWED_DETAIL_KEYS` (`outcome`); no PII, no counts with names.
- Backend tests run from `cadence-hr-hub/backend` with `python -m pytest <path> -v`. Frontend from `cadence-hr-hub/web` with `npx vitest run <path>`.
- Test DB is `hub_test` on localhost:15432 (served by the running cadence-hackathon postgres container; do not start a second one).

        entity_type="demo",
        entity_id=None,

    )

```typescript
export type UploadOutcome = { file: string; ok: boolean; created?: boolean; reason?: string };

/** Upload files through the existing single-file endpoint with a fixed concurrency cap. */
export async function uploadBatch(
  files: File[],
  upload: (file: File) => Promise<{ created: boolean }>,
  onProgress: (done: number, total: number) => void,
  concurrency = 4,
): Promise<UploadOutcome[]> {
  const outcomes: UploadOutcome[] = [];
  const queue = [...files];
  let done = 0;

  async function worker() {
    let file: File | undefined;
    while ((file = queue.shift()) !== undefined) {
      const current = file;
      try {
        const result = await upload(current);
        outcomes.push({ file: current.name, ok: true, created: result.created });
      } catch (reason) {
        outcomes.push({ file: current.name, ok: false, reason: reason instanceof Error ? reason.message : "upload_failed" });
      }
      done += 1;
      onProgress(done, files.length);
    }
  }

  await Promise.all(Array.from({ length: Math.min(concurrency, files.length) }, worker));
  return outcomes;
}
```

      if (poll.current) clearInterval(poll.current);
    };
  }, [processing, refresh]);

  const onPick = (list: FileList | null) => {
    setFiles(list ? [...list] : []);
    setOutcomes(null);
    setDone(0);
  };

  const onLoad = async () => {
    setUploading(true);
    setError(null);
    try {
      const result = await uploadBatch(files, uploadDocument, setDone);
      setOutcomes(result);
      refresh();
    } catch (reason) {
      setError(reason);
    } finally {
      setUploading(false);
    }
  };

  const onReset = async () => {
    setBusy(true);
    setError(null);
    try {
      await resetDemoData();
      setConfirming(false);
      setPhrase("");
      setFiles([]);
      setOutcomes(null);
      setDone(0);
      refresh();
    } catch (reason) {
      setError(reason);
    } finally {
      setBusy(false);
    }
  };

  const uploaded = outcomes?.filter((o) => o.ok) ?? [];
  const failed = outcomes?.filter((o) => !o.ok) ?? [];

  return (
    <>
      <PageHeader
        eyebrow="Demo / data"
        title="Demo data"
        sub="Operator controls for the walkthrough: reset to an empty operational database, then load the CV corpus through the normal ingestion path."
      />
      {error ? <ErrorState error={error} /> : null}
      <div className="grid grid-cols-[minmax(0,1fr)] gap-6 lg:grid-cols-2">
        <Panel as="section" className="p-5">
          <PanelHeader eyebrow="CV corpus" title="Load CVs" />
          <div className="space-y-3">
            <label className="block text-xs font-semibold">
              Corpus folder
              <input
                type="file"
                multiple
                // @ts-expect-error non-standard but supported by Chrome/Edge for folder pick
                webkitdirectory=""
                className={inputClass}
                onChange={(e) => onPick(e.target.files)}
              />
            </label>
            <label className="block text-xs font-semibold">
              Or pick files
              <input type="file" multiple className={inputClass} onChange={(e) => onPick(e.target.files)} />
            </label>
            {files.length > 0 ? <p className="text-sm">{files.length} files selected</p> : null}
            {uploading || done > 0 ? (
              <p className="text-sm" role="status">
                Uploaded {done} / {files.length}
              </p>
            ) : null}
            <Button variant="primary" busy={uploading} disabled={files.length === 0} onClick={onLoad}>
              Load CVs
            </Button>
            {outcomes ? (
              <div className="text-sm">
                <p>
                  Uploaded {uploaded.length} ({uploaded.filter((o) => o.created).length} new,{" "}
                  {uploaded.filter((o) => !o.created).length} already present) · Failed {failed.length}
                </p>
                {failed.length > 0 ? (
                  <ul className="mt-2 list-disc pl-5 text-bad-ink">
                    {failed.map((o) => (
                      <li key={o.file}>
                        {o.file}: {o.reason}
                      </li>
                    ))}
                  </ul>
                ) : null}
                <p className="mt-2 text-mut">Workers process in the background. Watch CV Intake for per-document review.</p>
              </div>
            ) : null}
          </div>
        </Panel>

        <Panel as="section" className="p-5">
          <PanelHeader eyebrow="Ingestion" title="Current data" />
          {status ? (
            <dl className="grid grid-cols-2 gap-x-6 gap-y-1 text-sm">
              <dt>Candidates</dt><dd className="tabular-nums">{status.candidates}</dd>
              <dt>Documents</dt><dd className="tabular-nums">{status.documents}</dd>
              <dt>Relationship reviews</dt><dd className="tabular-nums">{status.pending_reviews}</dd>
              <dt>OCR required</dt><dd className="tabular-nums">{status.needs_ocr}</dd>
              <dt>Quarantined</dt><dd className="tabular-nums">{status.quarantined}</dd>
              <dt>Hiring requests</dt><dd className="tabular-nums">{status.hiring_requests}</dd>
              <dt>Matches</dt><dd className="tabular-nums">{status.matches}</dd>
              <dt>Shortlisted</dt><dd className="tabular-nums">{status.shortlisted}</dd>
              <dt>Shares</dt><dd className="tabular-nums">{status.shares}</dd>
            </dl>
          ) : null}
          {status && status.processing > 0 ? <p className="mt-2 text-sm text-human-ink">Processing…</p> : null}
          <p className="mt-3 text-xs text-mut">
            Last reset: {status?.last_reset_at ? shortDateTime(status.last_reset_at) : "never"}
          </p>
        </Panel>

        <Panel as="section" className="border-bad-ink p-5">
          <PanelHeader eyebrow="Danger" title="Reset demo data" />
          <p className="text-sm text-mut">
            Permanently removes all candidate, CV, request, match, review and sharing data. Users and the audit trail are
            kept; the reset itself is recorded as a demo_reset audit event.
          </p>
          <Button variant="danger" className="mt-3" onClick={() => setConfirming(true)}>
            Reset demo data
          </Button>
        </Panel>
      </div>

      <ConfirmDialog
        open={confirming}
        title="Reset demo data?"
        body={
          <div className="space-y-3">
            <p>This permanently removes all candidate and demo data. This cannot be undone.</p>
            <label className="block text-xs font-semibold">
              Type RESET to confirm
              <input className={inputClass} value={phrase} onChange={(e) => setPhrase(e.target.value)} />
            </label>
          </div>
        }
        confirmLabel="Permanently wipe demo data"
        tone="danger"
        busy={busy}
        confirmDisabled={phrase !== "RESET"}
        onConfirm={onReset}
        onCancel={() => {
          setConfirming(false);
          setPhrase("");
        }}
      />
    </>
  );
}

```python
"""Demo-cycle validation: reset -> verify zero -> load cvs/ -> poll status -> reset -> verify zero.

Usage: python scripts/demo_cycle.py
Reads HUB_API (default http://localhost:8000) and HUB_RECRUITER_TOKEN (default dev-recruiter-token).
"""

import os
import time
import urllib.request
import json
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

API = os.environ.get("HUB_API", "http://localhost:8000")
TOKEN = os.environ.get("HUB_RECRUITER_TOKEN", "dev-recruiter-token")
CVS = Path(__file__).resolve().parents[1] / "cvs"


def call(method, path, body=None, raw=None, filename=None):
    headers = {"Authorization": f"Bearer {TOKEN}"}
    data = None
    if raw is not None:
        boundary = "boundary"
        payload = (
            f"--{boundary}\r\nContent-Disposition: form-data; name=\"file\"; filename=\"{filename}\"\r\n"
            "Content-Type: application/octet-stream\r\n\r\n"
        ).encode() + raw + f"\r\n--{boundary}--\r\n".encode()
        headers["Content-Type"] = f"multipart/form-data; boundary={boundary}"
        data = payload
    elif body is not None:
        headers["Content-Type"] = "application/json"
        data = json.dumps(body).encode()
    req = urllib.request.Request(API + path, data=data, headers=headers, method=method)
    with urllib.request.urlopen(req) as res:
        return json.loads(res.read())


def status():
    return call("GET", "/demo/status")


def main():
    started = time.perf_counter()
    call("POST", "/demo/reset", body={"confirm": "RESET"})
    zero = status()
    assert zero["candidates"] == 0 and zero["documents"] == 0, zero
    print(f"reset ok, zero verified ({time.perf_counter() - started:.1f}s)")

    files = [p for p in CVS.iterdir() if p.is_file()]
    print(f"{len(files)} files in cvs/")
    upload_start = time.perf_counter()
    created = failed = 0
    with ThreadPoolExecutor(max_workers=4) as pool:
        for result in pool.map(lambda p: _upload(p), files):
            if result == "created":
                created += 1
            elif result == "failed":
                failed += 1
    upload_s = time.perf_counter() - upload_start
    print(f"uploaded {len(files) - failed} ({created} new), failed {failed}, in {upload_s:.1f}s")

    poll_start = time.perf_counter()
    while True:
        s = status()
        if not s["processing"]:
            break
        time.sleep(3)
    process_s = time.perf_counter() - poll_start
    final = status()
    print(f"processing settled in {process_s:.1f}s")
    print(json.dumps(final, indent=2))

    call("POST", "/demo/reset", body={"confirm": "RESET"})
    zero_again = status()
    assert zero_again["candidates"] == 0 and zero_again["documents"] == 0, zero_again
    print(f"second reset ok, zero verified (total {time.perf_counter() - started:.1f}s)")


def _upload(path):
    try:
        return "created" if call("POST", "/documents", raw=path.read_bytes(), filename=path.name)["created"] else "duplicate"
    except Exception as exc:  # noqa: BLE001 - record and continue, like the browser loop
        print(f"FAILED {path.name}: {exc}")
        return "failed"


if __name__ == "__main__":
    main()
```
